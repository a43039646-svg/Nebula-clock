cmake_minimum_required(VERSION 3.21)
project(NebulaClock VERSION 2.0.0 LANGUAGES NONE)

install(DIRECTORY nebula_clock DESTINATION "lib/nebula-clock"
        FILES_MATCHING PATTERN "*.py" PATTERN "*.json")
install(PROGRAMS packaging/nebula-clock DESTINATION bin)
install(FILES LICENSE DESTINATION "share/licenses/nebula-clock")

set(CPACK_GENERATOR TGZ)
find_program(NEBULA_DPKG_DEB dpkg-deb)
if(NEBULA_DPKG_DEB)
    list(APPEND CPACK_GENERATOR DEB)
endif()
find_program(NEBULA_RPMBUILD rpmbuild)
if(NEBULA_RPMBUILD)
    list(APPEND CPACK_GENERATOR RPM)
endif()
set(CPACK_PACKAGE_NAME nebula-clock)
set(CPACK_PACKAGE_VENDOR "Nebula Project")
set(CPACK_PACKAGE_CONTACT "Nebula Project")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Customizable Linux terminal clock and distro theme utility")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Nebula Project")
set(CPACK_DEBIAN_PACKAGE_SECTION misc)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE all)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.10)")
set(CPACK_RPM_PACKAGE_LICENSE "EPL-2.0")
set(CPACK_RPM_PACKAGE_ARCHITECTURE noarch)
set(CPACK_RPM_PACKAGE_REQUIRES "python3 >= 3.10")
include(CPack)
